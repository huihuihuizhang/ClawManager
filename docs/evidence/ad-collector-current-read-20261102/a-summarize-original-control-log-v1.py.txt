import hashlib, json, pathlib, re, sys
source = pathlib.Path(sys.argv[1])
destination = pathlib.Path(sys.argv[2])
if destination.exists():
    raise SystemExit('refuse to replace existing evidence')
raw = source.read_bytes()
log = raw.decode('utf-8-sig')
if not re.search(r'^(?:ok\s+|FAIL\s+|panic: test timed out)', log, re.M):
    raise SystemExit('no completed Go package or natural timeout boundary')
markers = {
    'controller': 'original controller nested Authorize/SQLBind/key-state cost DATA ',
    'client': 'original control cumulative elapsed/count/failure DATA ',
}
observations = []
for line in log.splitlines():
    for kind, marker in markers.items():
        if marker not in line:
            continue
        operation, encoded = line.split(marker, 1)[1].split(' ', 1)
        payload = json.loads(encoded)
        item = {'kind': kind, 'operation': operation, 'data': payload}
        if kind == 'controller':
            item['key_check_stages_ranked_individually'] = [
                {'stage': name, **data}
                for name, data in sorted(payload['KeyCheck'].items(), key=lambda pair: pair[1]['ElapsedNanos'], reverse=True)
                if name != 'Authorize'
            ]
        observations.append(item)
result = {
    'Source': str(source.resolve()),
    'RawLogSHA256': hashlib.sha256(raw).hexdigest(),
    'Interpretation': 'Completed-call diagnostic DATA only. Authorize overlaps its child stages; no parent-plus-child sum. An empty or unavailable PFS sample does not prove absence of a blocker. Failed client/controller snapshots may end at different times.',
    'Observations': observations,
    'GateObservationLines': [line.strip() for line in log.splitlines() if 'original gate real3572 read-only lock DATA' in line or 'original gate lock DATA' in line or 'original gate refusal lock DATA unavailable' in line],
    'TerminalLines': [line.strip() for line in log.splitlines() if 'retained failed original SQL namespace' in line or 'original selection reached terminal' in line or line.startswith('--- FAIL:') or line.startswith('--- PASS:') or line.startswith('FAIL\t') or line.startswith('ok\t')],
}
with destination.open('x', encoding='utf-8') as output:
    json.dump(result, output, ensure_ascii=False, indent=2)
print(json.dumps({'observations': len(observations), 'gate_lines': len(result['GateObservationLines']), 'raw_sha256': result['RawLogSHA256']}))
